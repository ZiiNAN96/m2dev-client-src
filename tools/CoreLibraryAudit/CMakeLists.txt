# ZiiNAN: Explicitly opt-in measurement dependencies; never linked to the client.
set(CLIB_DEPS "${CMAKE_SOURCE_DIR}/build/clibx/deps" CACHE PATH "Prepared, pinned audit source archives")
set(CLIB_OZZ "${CLIB_DEPS}/ozz-animation-0.17.0")
set(CLIB_ACL "${CLIB_DEPS}/acl-2.1.0")
set(CLIB_RTM "${CLIB_DEPS}/rtm-d7982f2b2524feeb322f424b29cf43df30b5d5b7")
foreach(dep "${CLIB_OZZ}/CMakeLists.txt" "${CLIB_ACL}/includes/acl/version.h" "${CLIB_RTM}/includes/rtm/version.h")
    if(NOT EXISTS "${dep}")
        message(FATAL_ERROR "Run tools/CoreLibraryAudit/prepare_dependencies.py first; missing ${dep}")
    endif()
endforeach()
foreach(opt tools fbx gltf data samples howtos tests)
    set(ozz_build_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(ozz_build_msvc_rt_dll OFF CACHE BOOL "" FORCE)
add_subdirectory("${CLIB_OZZ}" "${CMAKE_CURRENT_BINARY_DIR}/ozz" EXCLUDE_FROM_ALL)
add_executable(CoreAnimationAudit AnimationAudit.cpp)
target_include_directories(CoreAnimationAudit SYSTEM PRIVATE "${CLIB_ACL}/includes" "${CLIB_RTM}/includes")
target_link_libraries(CoreAnimationAudit PRIVATE AssetRuntimeGR2 AssetRuntimeGlTF ozz_animation ozz_animation_offline)
target_compile_features(CoreAnimationAudit PRIVATE cxx_std_20)

set(CLIB_MESHOPT "${CLIB_DEPS}/meshoptimizer-0.25" CACHE PATH "Pinned meshoptimizer 0.25 source (same version as AssetTool)")
set(MESHOPT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MESHOPT_BUILD_GLTFPACK OFF CACHE BOOL "" FORCE)
set(MESHOPT_INSTALL OFF CACHE BOOL "" FORCE)
if(NOT TARGET meshoptimizer)
    add_subdirectory("${CLIB_MESHOPT}" "${CMAKE_CURRENT_BINARY_DIR}/meshoptimizer" EXCLUDE_FROM_ALL)
endif()
add_executable(CoreMeshAudit MeshAudit.cpp)
target_link_libraries(CoreMeshAudit PRIVATE AssetRuntimeGR2 AssetRuntimeGlTF meshoptimizer)

# The separately built upstream encoder archive also supplies the transcoder
# and Zstandard. Import only into this Windows measurement executable.
if(MSVC)
    set(CLIB_BASIS "${CLIB_DEPS}/basis_universal-2_50")
    add_library(CLibBasisProof STATIC IMPORTED)
    set_target_properties(CLibBasisProof PROPERTIES IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/build/clibx/basis-build/Release/basisu_encoder.lib")
    add_executable(CoreTextureAudit TextureAudit.cpp)
    target_include_directories(CoreTextureAudit PRIVATE "${CMAKE_SOURCE_DIR}/src")
    target_include_directories(CoreTextureAudit SYSTEM PRIVATE "${CLIB_BASIS}" "${CLIB_BASIS}/transcoder" "${CLIB_BASIS}/zstd" "${CMAKE_SOURCE_DIR}/extern/include")
    target_link_libraries(CoreTextureAudit PRIVATE CLibBasisProof)
    set_target_properties(CoreTextureAudit PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()

if(MSVC)
    target_compile_options(CoreAnimationAudit PRIVATE /O2 /fp:precise)
    target_compile_options(CoreMeshAudit PRIVATE /O2 /fp:precise)
else()
    target_compile_options(CoreAnimationAudit PRIVATE -O2 -fno-strict-aliasing)
    target_compile_options(CoreMeshAudit PRIVATE -O2)
endif()
