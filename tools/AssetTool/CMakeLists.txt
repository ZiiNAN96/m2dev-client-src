# Resolve the actual link closure, including private static-library dependencies.
function(ziinan_audit_runtime target)
    get_property(seen GLOBAL PROPERTY ZIINAN_AUDITED_TARGETS)
    if(target IN_LIST seen OR NOT TARGET "${target}")
        return()
    endif()
    set_property(GLOBAL APPEND PROPERTY ZIINAN_AUDITED_TARGETS "${target}")
    foreach(prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
        get_target_property(links "${target}" "${prop}")
        if("${links}" MATCHES "[Aa][Ss][Ss][Ii][Mm][Pp]|AssetToolCore|meshoptimizer")
            message(FATAL_ERROR "Offline dependency leaked into runtime target ${target}: ${links}")
        endif()
        foreach(link IN LISTS links)
            string(REGEX REPLACE "\\$<LINK_ONLY:([^>]+)>" "\\1" link "${link}")
            ziinan_audit_runtime("${link}")
        endforeach()
    endforeach()
endfunction()
foreach(runtime UserInterface AssetRuntime AssetRuntimeGlTF AssetRuntimeGR2 AssetRuntimeProviders M2RendererDiligent)
    ziinan_audit_runtime("${runtime}")
endforeach()
get_property(audited GLOBAL PROPERTY ZIINAN_AUDITED_TARGETS)
file(WRITE "${CMAKE_BINARY_DIR}/asset-tool-runtime-isolation.txt" "PASS: no Assimp/meshoptimizer/tool in runtime link closure\nTargets=${audited}\n")
