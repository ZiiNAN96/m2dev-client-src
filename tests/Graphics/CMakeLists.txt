find_package(Threads REQUIRED)
add_executable(GraphicsSettingsTest GraphicsSettingsTest.cpp)
target_link_libraries(GraphicsSettingsTest PRIVATE M2GraphicsSettings VegetationRuntime Threads::Threads)
foreach(mode Presets Custom Invalid LiveApply Vegetation Persistence WriteRestart ReadRestart)
    add_test(NAME Graphics.${mode} COMMAND GraphicsSettingsTest ${mode})
    set_tests_properties(Graphics.${mode} PROPERTIES LABELS "graphics;portable;fast;g0x" TIMEOUT 15)
endforeach()
set_tests_properties(Graphics.WriteRestart PROPERTIES FIXTURES_SETUP GraphicsRestart)
set_tests_properties(Graphics.ReadRestart PROPERTIES FIXTURES_REQUIRED GraphicsRestart)
if(M2_BUILD_RENDERER_TESTS)
    add_executable(GraphicsDiligentTest GraphicsDiligentTest.cpp)
    target_link_libraries(GraphicsDiligentTest PRIVATE M2GraphicsSettings M2RendererDiligent M2Platform)
    add_test(NAME Graphics.DiligentSnapshot COMMAND GraphicsDiligentTest)
    set_tests_properties(Graphics.DiligentSnapshot PROPERTIES LABELS "graphics;gpu;fast;g0x" TIMEOUT 30)
endif()
